\documentclass[tikz,border=2mm]{standalone}

\usepackage{tikz}
\usepackage{pgfplots}
\usepackage{mathrsfs}
\usepgfplotslibrary{groupplots}
\usetikzlibrary{calc}
\pgfplotsset{compat=1.18}

\newcommand\sE{{\mathscr E}}
\newcommand\sR{{\mathscr R}}

\pgfmathsetmacro{\opacity}{50}
\colorlet{lightred}{red!\opacity}
\colorlet{lightgreen}{green!\opacity}
\definecolor{lightblue}{RGB}{144,202,249}

\begin{document}

\pgfplotsset{compat=1.17}

\pgfmathsetmacro{\pThree}{(3-sqrt(5))/2}
\pgfmathsetmacro{\gThree}{(5-sqrt(5))/4}
\pgfmathsetmacro{\HpThree}{0.9594187282}
\pgfmathsetmacro{\slopeThree}{(1/(1+\pThree))*\HpThree}
\pgfmathsetmacro{\coefThree}{0.5}

\pgfmathsetmacro{\pFour}{0.3176721962}
\pgfmathsetmacro{\gFour}{(1+2*\pFour)/3}
\pgfmathsetmacro{\HpFour}{0.9018320714}
\pgfmathsetmacro{\slopeFour}{(2/(1+2*\pFour))*\HpFour}
\pgfmathsetmacro{\coefFour}{2/3}

\pgfmathsetmacro{\pFive}{0.2755080410}
\pgfmathsetmacro{\gFive}{(1+3*\pFive)/4}
\pgfmathsetmacro{\HpFive}{0.8492577652}
\pgfmathsetmacro{\slopeFive}{(3/(1+3*\pFive))*\HpFive}
\pgfmathsetmacro{\coefFive}{3/4}

\newcommand{\scale}{1.0}
\newcommand{\curvewidth}{very thick}

\begin{tikzpicture}[scale=\scale]
	\pgfplotsset{set layers}
	\begin{groupplot}[
		group style={
			group size=2 by 1,
			horizontal sep=1.2cm,
			vertical sep=0cm
		},
		width=7.8cm,
		height=5.55cm,
		grid=both,
		major grid style={solid,gray!30},
		every axis plot/.append style={\curvewidth},
		xticklabel style={/pgf/number format/fixed},
		yticklabel style={/pgf/number format/fixed}
	]

	\nextgroupplot[
		xmin=0, xmax=1.05,
		ymin=0, ymax=0.82,
		xlabel={\(\gamma\)},
		axis x line=bottom,
		axis y line=middle,
		xtick={0,0.2,0.4,0.6,0.8,1},
		ytick={0,0.2,0.4,0.6,0.8},
		label style={font=\small},
		tick label style={font=\small},
		line cap=round,
		legend style={
			at={(rel axis cs:0.03,0.97)},
			anchor=north west,
			draw=none,
			fill=white,
			fill opacity=0.85,
			text opacity=1,
			font=\scriptsize,
			inner sep=1.2pt,
			legend columns=1
		},
	]

	\addplot[smooth,domain=0:\gThree,samples=60,color=lightred] {\slopeThree*x};
	\addlegendentry{$\sE_3$}
	\addplot[smooth,domain=\gThree:1,samples=140,color=lightred,forget plot]
		{\coefThree * ( -((2*x-1))*log2((2*x-1)) - (1-(2*x-1))*log2(1-(2*x-1)) )};

	\addplot[smooth,domain=0:\gFour,samples=60,color=lightgreen] {\slopeFour*x};
	\addlegendentry{$\sE_4$}
	\addplot[smooth,domain=\gFour:1,samples=160,color=lightgreen,forget plot]
		{\coefFour * ( -(((3*x-1)/2))*log2(((3*x-1)/2)) - (1-((3*x-1)/2))*log2(1-((3*x-1)/2)) )};

	\addplot[smooth,domain=0:\gFive,samples=60,color=lightblue] {\slopeFive*x};
	\addlegendentry{$\sE_5$}
	\addplot[smooth,domain=\gFive:1,samples=180,color=lightblue,forget plot]
		{\coefFive * ( -(((4*x-1)/3))*log2(((4*x-1)/3)) - (1-((4*x-1)/3))*log2(1-((4*x-1)/3)) )};

	\nextgroupplot[
		xmin=0, xmax=1.05,
		ymin=0, ymax=0.82,
		xlabel={$p$},
		axis x line=bottom,
		axis y line=middle,
		xtick={0,0.2,0.4,0.6,0.8,1},
		ytick={0,0.2,0.4,0.6,0.8},
		xlabel near ticks,
		label style={font=\small},
		tick label style={font=\small},
		line cap=round,
		legend style={
			at={(rel axis cs:0.97,0.97)},
			anchor=north east,
			draw=none,
			fill=white,
			fill opacity=0.85,
			text opacity=1,
			font=\scriptsize,
			inner sep=1.2pt,
			legend columns=1
		},
	]

	\addplot[smooth,domain=0:\pThree,samples=80,color=lightred] {-log2(1-x)};
	\addlegendentry{$\sR_3$}
	\addplot[smooth,domain=\pThree:1,samples=140,color=lightred,forget plot] {-(1/2)*log2(x)};

	\addplot[smooth,domain=0:\pFour,samples=80,color=lightgreen] {-log2(1-x)};
	\addlegendentry{$\sR_4$}
	\addplot[smooth,domain=\pFour:1,samples=140,color=lightgreen,forget plot] {-(1/3)*log2(x)};

	\addplot[smooth,domain=0:\pFive,samples=80,color=lightblue] {-log2(1-x)};
	\addlegendentry{$\sR_5$}
	\addplot[smooth,domain=\pFive:1,samples=140,color=lightblue,forget plot] {-(1/4)*log2(x)};

	\end{groupplot}
\end{tikzpicture}

\end{document}
